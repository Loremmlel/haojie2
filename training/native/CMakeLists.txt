cmake_minimum_required(VERSION 3.24)
project(haojie_train LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 可使用独立 LibTorch，也可复用已安装包中的 C++ 库；不链接 Python。
if(NOT LIBTORCH AND DEFINED ENV{LIBTORCH})
  set(LIBTORCH "$ENV{LIBTORCH}")
endif()
if(NOT LIBTORCH)
  message(FATAL_ERROR "Set LIBTORCH to an extracted LibTorch directory")
endif()
find_package(Threads REQUIRED)
add_executable(haojie-train src/main.cpp src/model.cpp src/runtime.cpp)
target_include_directories(haojie-train PRIVATE "${LIBTORCH}/include" "${LIBTORCH}/include/torch/csrc/api/include" src)
target_link_directories(haojie-train PRIVATE "${LIBTORCH}/lib")
target_link_libraries(haojie-train PRIVATE torch torch_cpu c10 Threads::Threads)
if(EXISTS "${LIBTORCH}/lib/torch_cuda.lib" OR EXISTS "${LIBTORCH}/lib/libtorch_cuda.so")
  target_compile_definitions(haojie-train PRIVATE HJ_CUDA)
  target_link_libraries(haojie-train PRIVATE torch_cuda c10_cuda)
endif()
if(WIN32)
  target_compile_options(haojie-train PRIVATE /utf-8 /bigobj /O2 /wd4267 /wd4244)
  target_link_libraries(haojie-train PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/native-target/release/haojie_native.dll.lib")
  if(EXISTS "${LIBTORCH}/lib/torch_cuda.lib")
    target_link_options(haojie-train PRIVATE "/INCLUDE:?warp_size@cuda@at@@YAHXZ")
  endif()
else()
  target_compile_options(haojie-train PRIVATE -O3)
  if(EXISTS "${LIBTORCH}/lib/libtorch_cuda.so")
    # CUDA 后端依靠共享库初始化注册，保留它与 CPU 调度器之间的链接。
    target_link_options(haojie-train PRIVATE "-Wl,--no-as-needed")
  endif()
  target_link_libraries(haojie-train PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/native-target/release/libhaojie_native.so" ${CMAKE_DL_LIBS})
  set_target_properties(haojie-train PROPERTIES BUILD_RPATH "${LIBTORCH}/lib;${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/native-target/release")
endif()
